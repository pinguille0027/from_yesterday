# =====================================================================
# FIX (ronda 2): "subtract_variable" / "divide_variable" NO EXISTEN como
# efectos independientes (confirmado: "Unknown effect subtract_variable/
# divide_variable"). La forma real de hacer aritmética entre variables
# en Vic3 es anidar un bloque de "valor" con los operadores add/subtract/
# multiply/divide/min/max DENTRO del campo "value" de set_variable,
# exactamente como en un script value con nombre. Ejemplo real de la
# wiki:
#   first_example = { add = 6 multiply = { value = 3 divide = 2 } max = 10 subtract = 3 }
# Aplicamos ese mismo patrón aquí, y aprovechamos "min"/"max" para
# sustituir también a clamp_variable (que no hemos podido confirmar).
# =====================================================================

on_battle_started = {
	on_actions = {
		fy_on_battle_started
	}
}

on_battle_ended = {
	on_actions = {
		fy_on_battle_ended
	}
	events = {
		fy_disasters.1
	}
}

fy_on_battle_started = {
	effect = {
		hidden_effect = {
			set_variable = {
				name = fy_battle_start_own_army
				value = army_size_including_conscripts
			}
			if = {
				limit = { exists = scope:enemy_country }
				scope:enemy_country = {
					save_scope_as = fy_current_enemy
				}
				set_variable = {
					name = fy_battle_start_enemy_army
					value = scope:enemy_country.army_size_including_conscripts
				}
			}
		}
	}
}

fy_on_battle_ended = {
	effect = {
		hidden_effect = {
			if = {
				limit = {
					exists = var:fy_battle_start_own_army
					exists = var:fy_battle_start_enemy_army
					exists = scope:fy_current_enemy
				}

				set_variable = { name = fy_own_army_after value = army_size_including_conscripts }
				set_variable = { name = fy_enemy_army_after value = scope:fy_current_enemy.army_size_including_conscripts }

				# bajas propias = max(0, ejercito_antes - ejercito_despues)
				set_variable = {
					name = fy_own_losses
					value = {
						value = var:fy_battle_start_own_army
						subtract = var:fy_own_army_after
						min = 0
					}
				}

				set_variable = {
					name = fy_enemy_losses
					value = {
						value = var:fy_battle_start_enemy_army
						subtract = var:fy_enemy_army_after
						min = 0
					}
				}

				set_variable = {
					name = fy_own_loss_pct
					value = {
						value = var:fy_own_losses
						divide = var:fy_battle_start_own_army
					}
				}

				set_variable = {
					name = fy_enemy_loss_pct
					value = {
						value = var:fy_enemy_losses
						divide = var:fy_battle_start_enemy_army
					}
				}

				set_variable = {
					name = fy_casualty_gap
					value = {
						value = var:fy_own_loss_pct
						subtract = var:fy_enemy_loss_pct
					}
				}
			}
		}
	}
}
